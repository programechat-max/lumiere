def upgrade():
    conn = op.get_bind()
    existing = _existing_tables(conn)          # mevcut tablolar
    for table_name, columns, indexes in TABLES:
        if table_name in existing:
            continue                            # idempotent: dokunma

        for index_name, index_cols in indexes:
            op.create_index(op.f(index_name), table_name, index_cols)

▸ D2: food_db — yarım kalmış diyet filtresi

_diet_tag_filters() gövdesi banned = set() satırında bitiyor, ardından bir sonraki fonksiyonun satırı araya sıkışmıştı (sökülmüş kod). Diyet kısıtları hiç uygulanmıyordu. Yerine tam işlevsel _diet_allows() yazıldı:

def _diet_allows(dietary_notes, tags) -> bool:
    notes = _norm(dietary_notes)
    if not notes: return True                      # boş not: her şeye izin
    if "vegan" in notes:
        if "vegan" not in tags: return False
    elif "vejetaryen" in notes or "vegetarian" in notes:
        if "vegan" not in tags and "vegetarian" not in tags: return False
    if "gluten" in notes or "glutensiz" in notes:
        if "gluten_free" not in tags: return False
    if "laktoz" in notes or "süt" in notes:
        if "lactose_free" not in tags and "vegan" not in tags: return False
    return True
Örnek: search_candidate_foods(dietary_notes="vegan") artık yalnızca vegan tag'li besinleri döndürür ve havuz yine dolu kalır (test 15/15 içinde doğrulanıyor).

▸ D3: verify_and_fix_meal_plan — gramaj/parantez eşleşme hatası

Doğrulama, malzeme metnini olduğu gibi ("150g tavuk göğsü (ızgara)") resolve_food_name()'e gönderiyordu; hiçbir tablo eşleşmesi bulunamıyor, düzeltme hiç tetiklenmiyordu. İki adımlı çözüm uygulandı:

def _clean_ingredient_name(text: str) -> str:
    cleaned = (text or "").lower()
    cleaned = re.sub(r"^\s*\d+(?:[.,]\d+)?\s*(?:g|gram)\b", " ", cleaned)  # basit gramaj
    cleaned = re.sub(r"[\(\[].*?[\)\]]", " ", cleaned)                      # parantez notu
    cleaned = re.sub(r"\b(?:ve|with|ile)\b", " ", cleaned)                  # baglaclar
    return " ".join(cleaned.split())

# "150g tavuk göğsü (ızgara)"  -->  "tavuk göğsü"  -->  TABLODA EŞLEŞİR
# porsiyon gramajı yine de orijinal metinden: _extract_portion_g("150g ...") = 150.0

Ek olarak eski "porsiyon ölçekle" mantığı kaldırıldı. Artık AI'nın beyan ettiği yanlış makrolar, tablodan hesaplanan doğru değerlerle değiştiriliyor — kullanıcının tarifi ve gramajı bozulmadan:

if matched_any and item.get("calories"):

    if declared > 0 and computed["calories"] > 0:

if matched_any:
    # Beyan edilen makrolar yerine tablodan hesaplanan DOĞRU değerleri yaz.

▸ D4: resolve_food_name — kelime sınırına duyarlı eşleşme

ilike "%yumurt%" sorgusu Yumurta (haşlanmış)'ın yanı sıra menemen yumurtalı alias'ını da döndürüyor, sıralamada yanlış kayıt seçilebiliyordu. Artık önce kelime sınırına duyarlı (regex) tam kelime eşleşmesi aranıyor, bulunamazsa içerir-mantığına düşülüyor:

def _word_match(q: str, item: FoodItem) -> bool:
    pattern = re.compile(rf"(?<![a-zçğıöşü0-9]){re.escape(q)}(?![a-zçğıöşü0-9])")
    return bool(pattern.search(_norm(item.name)) or pattern.search(_norm(item.aliases or "")))

# "yumurta"        -> "_word_match" yalnizca "Yumurta (haşlanmış)" secer
#                    (menemen YUMURTAlı alias'i kelime sinirina takilmaz)
# "tavuğu"         -> once tam metin, sonra kelime kademeli ("tavuk"),
#                    en son stem: cekim eki kirpilir -> Tavuk Göğsü eslesir

Dokunmatik swipe ile geçiş (onTouchStart/onTouchEnd), her sekme değişiminde yukarı kaydırma ve oturum süresi dolunca lumiere:session-expired ile login'e dönüş davranışları eklendi.
